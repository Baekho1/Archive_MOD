						ROOT = { has_idea = PASTA_shock_to_daejeon }
					}

						set_country_flag = ITA_human_fascism_lock_trigger_flag
						country_event = PASTA_event.8
					}
				}
				else_if = { # 파시즘 bop 보유시 격발
					limit = {
						ROOT = {
							has_power_balance = { id = ITA_power_balance }
							has_government = fascism
							NOT = { has_country_flag = ITA_kvt_bop_modifier_lock_trigger_flag }
						}
					}
					ROOT = {
						add_power_balance_modifier = {
							id = ITA_power_balance
							modifier = ITA_power_balance_aliy_in_kvt_country
						}
					}
				}
			}
			else_if = {
				limit = {
					Kivotos_tag = yes
