on_actions = {

	on_startup = {
		effect = {
			every_country = {
				if = {
					limit = { tag = GEH }

					#선도부 풍기
					geh_discipline_system_setting_effect = yes
					add_dynamic_modifier = {
						modifier = geh_discipline_idea
					}

					# AI 선도부 루트 설정
					if = {
						limit = {
							GEM = { is_ai = yes }
							tag = GEH
						}
						random_list = {
							10 = {
								set_variable = {
									GEH_route_random = 0 # 히나 독재정
								}
							}
							10 = {
								set_variable = {
									GEH_route_random = 1 # 이오리 루트
								}
							}
							10 = {
								set_variable = {
									GEH_route_random = 2 # 아코 루트
								}
							}
							10 = {
								set_variable = {
									GEH_route_random = 3 # 기본 루트
								}
							}
						}
					}
					else_if = {
						limit = {
							GEM = { is_ai = no } #만마전이 플레이어임
							tag = GEH
						}
						random_list = {
							10 = {
								set_variable = {
									GEH_route_random = 0 # 히나 독재정
								}
							}
							10 = {
								set_variable = {
									GEH_route_random = 2 # 아코 루트
								}
							}
						}
					}
				}
			}
		}
	}

}
